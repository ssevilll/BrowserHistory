cmake_minimum_required(VERSION 3.20)
project(BrowserHistoryManagement VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/W4)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(browser_history_core src/BrowserHistory.cpp)
target_include_directories(browser_history_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
add_executable(browser_history src/main.cpp)
target_link_libraries(browser_history PRIVATE browser_history_core)
add_executable(browser_history_native_tests tests/native/BrowserHistoryNativeTests.cpp)
target_link_libraries(browser_history_native_tests PRIVATE browser_history_core)
enable_testing()
add_test(NAME BrowserHistoryNativeTests COMMAND browser_history_native_tests)
